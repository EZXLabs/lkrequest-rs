cmake_minimum_required(VERSION 3.20)
project(tai_boringssl_peer LANGUAGES CXX)
set(BORINGSSL_SOURCE_DIR "" CACHE PATH "Pinned BoringSSL source checkout")
set(BORINGSSL_BUILD_DIR "" CACHE PATH "BoringSSL build containing ssl and crypto libraries")
find_path(BORINGSSL_INCLUDE openssl/ssl.h PATHS "${BORINGSSL_SOURCE_DIR}/include" NO_DEFAULT_PATH REQUIRED)
find_library(BORINGSSL_SSL NAMES ssl PATHS "${BORINGSSL_BUILD_DIR}/Release" "${BORINGSSL_BUILD_DIR}" "${BORINGSSL_BUILD_DIR}/ssl" NO_DEFAULT_PATH REQUIRED)
find_library(BORINGSSL_CRYPTO NAMES crypto PATHS "${BORINGSSL_BUILD_DIR}/Release" "${BORINGSSL_BUILD_DIR}" "${BORINGSSL_BUILD_DIR}/crypto" NO_DEFAULT_PATH REQUIRED)
add_executable(tai_boringssl_peer server.cc)
target_compile_features(tai_boringssl_peer PRIVATE cxx_std_17)
target_include_directories(tai_boringssl_peer PRIVATE "${BORINGSSL_INCLUDE}")
target_link_libraries(tai_boringssl_peer PRIVATE "${BORINGSSL_SSL}" "${BORINGSSL_CRYPTO}")
if(WIN32)
  target_link_libraries(tai_boringssl_peer PRIVATE ws2_32 advapi32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(tai_boringssl_peer PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endif()
